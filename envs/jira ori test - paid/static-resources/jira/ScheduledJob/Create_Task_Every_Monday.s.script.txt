// Replace TEST with the project key you'd like to create a release in
def projectKey = 'TEST'

// Here we get the issue id for 'Task' to use later
def taskType = get('/rest/api/2/issuetype').asObject(List).body.find { it['name'] == 'Task' }['id']

// Here we tell jira to create the Task as specified
post('/rest/api/2/issue')
        .header('Content-Type', 'application/json')
        .body(
        [
                fields: [
                        summary    : 'My tasks to do every Monday',
                        description: "Don't forget to do these!.",
                        project    : [
                                key: projectKey
                        ],
                        issuetype  : [
                                id: taskType
                        ]
                ]
        ])
        .asString().body
